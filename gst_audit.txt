class Product(models.Model):
    category = models.ForeignKey(
        "categories.Category",
        on_delete=models.PROTECT,
        related_name="products",
    )
    name = models.CharField(max_length=200, db_index=True)
    slug = models.SlugField(max_length=220, unique=True, blank=True)
    description = models.TextField(blank=True)
    price = models.DecimalField(
        "original price",
        max_digits=10,
        decimal_places=2,
        validators=[MinValueValidator(Decimal("0.01"))],
    )
    discount_price = models.DecimalField(
        max_digits=10,
        decimal_places=2,
        null=True,
        blank=True,
    )
    stock = models.PositiveIntegerField(default=0)
    low_stock_threshold = models.PositiveIntegerField(default=5)
    is_active = models.BooleanField(default=True)
    is_featured = models.BooleanField(default=False)
    sold_count = models.PositiveIntegerField(default=0)
    avg_rating = models.DecimalField(max_digits=3, decimal_places=2, default=0)
    review_count = models.PositiveIntegerField(default=0)
    created_at = models.DateTimeField(auto_now_add=True, db_index=True)
    updated_at = models.DateTimeField(auto_now=True)
       # GST Configuration
    hsn_code = models.CharField(
        max_length=8,
        blank=True,
        help_text="Enter the applicable HSN code.",
    )

    gst_rate = models.DecimalField(
        max_digits=5,
        decimal_places=2,
        null=True,
        blank=True,
        validators=[
            MinValueValidator(Decimal("0")),
            MaxValueValidator(Decimal("100")),
        ],
        help_text="Product GST rate in percentage.",
    )
    class Meta:
        ordering = ["-created_at"]
        constraints = [
            models.CheckConstraint(
                check=Q(discount_price__isnull=True)
                | Q(discount_price__lt=F("price")),
                name="discount_less_than_price",
            )
        ]

    def save(self, *args, **kwargs):
        if not self.slug:
            base = slugify(self.name)[:200]
            self.slug = base
            n = 1

            while Product.objects.filter(slug=self.slug).exclude(pk=self.pk).exists():
                n += 1
                self.slug = f"{base}-{n}"

        super().save(*args, **kwargs)

    @property
    def effective_price(self):
        return (
            self.discount_price
            if self.discount_price is not None
            else self.price
        )

    @property
    def discount_percent(self):
        if self.discount_price is None:
            return 0

        return int(
            round((self.price - self.discount_price) / self.price * 100)
        )

    @property
    def in_stock(self):
        return self.stock > 0

    @property
    def is_low_stock(self):
        return 0 < self.stock <= self.low_stock_threshold

    def __str__(self):
        return self.name

CENT = Decimal("0.01")
HUNDRED = Decimal("100")


def _money(value):
    """Round a monetary value to two decimal places."""
    return Decimal(value).quantize(CENT, rounding=ROUND_HALF_UP)


def delivery_charge_for(amount):
    amount = _money(amount)
    return (
        Decimal("0.00")
        if amount >= settings.FREE_DELIVERY_THRESHOLD
        else _money(settings.DELIVERY_CHARGE)
    )


def _cart_lines(user, lock):
    """Validate cart products and live stock."""
    cart_items = list(get_cart(user).items.all())

    if not cart_items:
        raise ValidationError({"cart": "Your cart is empty."})

    qs = Product.objects.filter(
        id__in=[item.product_id for item in cart_items]
    ).order_by("id")

    products = {
        product.id: product
        for product in (qs.select_for_update() if lock else qs)
    }

    lines = []

    for item in cart_items:
        product = products.get(item.product_id)

        if product is None or not product.is_active:
            raise ValidationError({
                "cart": "A product in your cart is no longer available."
            })

        if item.quantity > product.stock:
            raise ValidationError({
                "cart": (
                    f"Only {product.stock} unit(s) of "
                    f"'{product.name}' left in stock."
                )
            })

        lines.append((product, item.quantity))

    return lines


def _allocate_amount(total, weights):
    """
    Allocate a monetary amount proportionally.
    The final line receives the rounding remainder so allocations
    always add up exactly to the original amount.
    """
    total = _money(total)

    if not weights:
        return []

    total_weight = sum(weights, Decimal("0"))

    if total_weight <= 0:
        if total == 0:
            return [Decimal("0.00") for _ in weights]

        raise ValidationError({
            "cart": "Unable to distribute the order discount or delivery fee."
        })

    allocations = []
    allocated = Decimal("0.00")

    for index, weight in enumerate(weights):
        if index == len(weights) - 1:
            value = _money(total - allocated)
        else:
            value = _money(total * weight / total_weight)
            allocated += value

        allocations.append(value)

    return allocations


def _tax_type_for_state(state):
    """VELORA is registered in Tamil Nadu."""
    normalized = (state or "").strip().casefold()
    normalized = " ".join(normalized.replace(".", "").split())

    if not normalized:
        raise ValidationError({
            "address": "The delivery address must contain a state."
        })

    tamil_nadu_names = {
        "tamil nadu",
        "tamilnadu",
        "tn",
    }

    return (
        TaxType.INTRA_STATE
        if normalized in tamil_nadu_names
        else TaxType.INTER_STATE
    )


def _totals(lines, user, coupon_code, lock, delivery_state=None):
    """
    Calculate product discount, coupon discount, delivery allocation,
    taxable value and GST for every cart line.

    Product prices are GST-exclusive.
    """

    if not lines:
        raise ValidationError({"cart": "Your cart is empty."})

    original_items_total = _money(sum(
        (product.price * quantity for product, quantity in lines),
        Decimal("0.00"),
    ))

    line_amounts = [
        _money(product.effective_price * quantity)
        for product, quantity in lines
    ]

    items_total = _money(sum(line_amounts, Decimal("0.00")))
    product_discount = _money(original_items_total - items_total)

    coupon = None
    coupon_discount = Decimal("0.00")

    if coupon_code:
        coupon, coupon_discount = validate_coupon(
            coupon_code,
            user,
            items_total,
            lock=lock,
        )
        coupon_discount = _money(coupon_discount)

    if coupon_discount < 0 or coupon_discount > items_total:
        raise ValidationError({
            "coupon": "The calculated coupon discount is invalid."
        })

    delivery = delivery_charge_for(items_total - coupon_discount)

    # Allocate coupon discount proportionally across product lines.
    coupon_allocations = _allocate_amount(
        coupon_discount,
        line_amounts,
    )

    discounted_line_amounts = [
        _money(line_amount - coupon_share)
        for line_amount, coupon_share in zip(
            line_amounts,
            coupon_allocations,
        )
    ]

    # Delivery is included in the taxable value and allocated across
    # the discounted product lines.
    delivery_weights = discounted_line_amounts

    if sum(delivery_weights, Decimal("0")) <= 0:
        delivery_weights = line_amounts

    delivery_allocations = _allocate_amount(
        delivery,
        delivery_weights,
    )

    tax_type = (
        _tax_type_for_state(delivery_state)
        if delivery_state is not None
        else None
    )

    item_tax_details = []

    taxable_amount = Decimal("0.00")
    gst_amount = Decimal("0.00")
    cgst_amount = Decimal("0.00")
    sgst_amount = Decimal("0.00")
    igst_amount = Decimal("0.00")

    for index, (product, quantity) in enumerate(lines):
        if product.gst_rate is None:
            raise ValidationError({
                "gst": (
                    f"GST rate is missing for '{product.name}'. "
                    "Update its GST rate in Django Admin before checkout."
                )
            })

        gst_rate = Decimal(product.gst_rate)

        if gst_rate < 0 or gst_rate > 100:
            raise ValidationError({
                "gst": f"Invalid GST rate configured for '{product.name}'."
            })

        taxable_value = _money(
            discounted_line_amounts[index]
            + delivery_allocations[index]
        )

        line_gst = _money(taxable_value * gst_rate / HUNDRED)

        line_cgst = Decimal("0.00")
        line_sgst = Decimal("0.00")
        line_igst = Decimal("0.00")

        if tax_type == TaxType.INTRA_STATE:
            # Split the rounded GST amount, ensuring both components
            # add up exactly to the line GST amount.
            line_cgst = _money(line_gst / Decimal("2"))
            line_sgst = _money(line_gst - line_cgst)

        elif tax_type == TaxType.INTER_STATE:
            line_igst = line_gst

        taxable_amount += taxable_value
        gst_amount += line_gst
        cgst_amount += line_cgst
        sgst_amount += line_sgst
        igst_amount += line_igst

        item_tax_details.append({
            "product": product,
            "quantity": quantity,
            "line_amount": line_amounts[index],
            "product_discount": _money(
                (product.price * quantity) - line_amounts[index]
            ),
            "coupon_discount": coupon_allocations[index],
            "delivery_allocation": delivery_allocations[index],
            "taxable_value": taxable_value,
            "hsn_code": product.hsn_code or "",
            "gst_rate": gst_rate,
            "gst_amount": line_gst,
            "cgst_amount": line_cgst,
            "sgst_amount": line_sgst,
            "igst_amount": line_igst,
        })

    taxable_amount = _money(taxable_amount)
    gst_amount = _money(gst_amount)
    cgst_amount = _money(cgst_amount)
    sgst_amount = _money(sgst_amount)
    igst_amount = _money(igst_amount)

    total_amount = _money(taxable_amount + gst_amount)

    return {
        "original_items_total": original_items_total,
        "items_total": items_total,
        "product_discount": product_discount,
        "coupon": coupon,
        "coupon_discount": coupon_discount,
        "delivery_charge": delivery,
        "taxable_amount": taxable_amount,
        "gst_amount": gst_amount,
        "cgst_amount": cgst_amount,
        "sgst_amount": sgst_amount,
        "igst_amount": igst_amount,
        "tax_type": tax_type,
        "total_amount": total_amount,
        "item_tax_details": item_tax_details,
    }


def checkout_summary(user, coupon_code=None, address_id=None):
    """
    Calculate checkout preview.

    Pass address_id to receive the correct CGST/SGST or IGST breakdown.
    Without an address, the overall GST amount is still calculated,
    but the state-wise tax split is left unspecified.
    """
    lines = _cart_lines(user, lock=False)

    delivery_state = None

    if address_id:
        address = Address.objects.filter(
            user=user,
            pk=address_id,
        ).first()

        if not address:
            raise ValidationError({
                "address_id": "Select a valid delivery address."
            })

        delivery_state = address.state

    totals = _totals(
        lines,
        user,
        coupon_code,
        lock=False,
        delivery_state=delivery_state,
    )

    return {
        "original_items_total": totals["original_items_total"],
        "items_total": totals["items_total"],
        "product_discount": totals["product_discount"],
        "coupon_code": (
            totals["coupon"].code if totals["coupon"] else ""
        ),
        "coupon_discount": totals["coupon_discount"],
        "delivery_charge": totals["delivery_charge"],
        "taxable_amount": totals["taxable_amount"],
        "gst_amount": totals["gst_amount"],
        "cgst_amount": totals["cgst_amount"],
        "sgst_amount": totals["sgst_amount"],
        "igst_amount": totals["igst_amount"],
        "tax_type": totals["tax_type"],
        "total_amount": totals["total_amount"],
        "item_count": sum(quantity for _, quantity in lines),
    }

@transaction.atomic
def place_order(user, address_id, payment_method, coupon_code=None):
    lines = _cart_lines(user, lock=True)

    address = Address.objects.filter(
        user=user,
        pk=address_id,
    ).first()

    if not address:
        raise ValidationError({
            "address_id": "Select a valid delivery address."
        })

    totals = _totals(
        lines,
        user,
        coupon_code,
        lock=True,
        delivery_state=address.state,
    )

    cod = payment_method == PaymentMethod.COD

    order = Order.objects.create(
        user=user,
        status=(
            OrderStatus.PLACED
            if cod
            else OrderStatus.PENDING_PAYMENT
        ),
        payment_method=payment_method,
        payment_status=PaymentStatus.PENDING,
        address=address,
        shipping_name=address.full_name,
        shipping_phone=address.phone,
        shipping_address=", ".join(filter(None, [
            address.house_number,
            address.street,
            address.city,
            address.state,
            address.postal_code,
            (
                f"Note: {address.delivery_instructions}"
                if address.delivery_instructions
                else ""
            ),
        ])),

        items_total=totals["items_total"],
        original_items_total=totals["original_items_total"],
        product_discount=totals["product_discount"],

        coupon=totals["coupon"],
        coupon_code=(
            totals["coupon"].code if totals["coupon"] else ""
        ),
        coupon_discount=totals["coupon_discount"],

        delivery_charge=totals["delivery_charge"],
        taxable_amount=totals["taxable_amount"],
        gst_amount=totals["gst_amount"],
        cgst_amount=totals["cgst_amount"],
        sgst_amount=totals["sgst_amount"],
        igst_amount=totals["igst_amount"],
        tax_type=totals["tax_type"],

        total_amount=totals["total_amount"],
        estimated_delivery=(
            timezone.localdate()
            + timedelta(days=settings.ESTIMATED_DELIVERY_DAYS)
        ),
    )

    for detail in totals["item_tax_details"]:
        product = detail["product"]
        quantity = detail["quantity"]

        OrderItem.objects.create(
            order=order,
            product=product,
            product_name=product.name,
            original_price=product.price,
            unit_price=product.effective_price,
            quantity=quantity,
            line_total=detail["line_amount"],

            # Snapshot GST information at order time.
            hsn_code=detail["hsn_code"],
            gst_rate=detail["gst_rate"],
            product_discount=detail["product_discount"],
            coupon_discount=detail["coupon_discount"],
            taxable_value=detail["taxable_value"],
            gst_amount=detail["gst_amount"],
            cgst_amount=detail["cgst_amount"],
            sgst_amount=detail["sgst_amount"],
            igst_amount=detail["igst_amount"],
        )

        product.stock -= quantity
        product.sold_count += quantity
        product.save(update_fields=[
            "stock",
            "sold_count",
            "updated_at",
        ])

        log_stock(product, -quantity, "Order placed", order)

    if totals["coupon"]:
        Coupon.objects.filter(
            pk=totals["coupon"].pk
        ).update(used_count=F("used_count") + 1)

    if cod:
        from payments.models import Payment, PaymentRecordStatus

        Payment.objects.create(
            order=order,
            method=PaymentMethod.COD,
            amount=totals["total_amount"],
            status=PaymentRecordStatus.CREATED,
        )

        _record(
            order,
            OrderStatus.PLACED,
            "Order placed (cash on delivery)",
        )

        notify(
            user,
            "Order placed",
            f"Your order {order.order_number} has been placed.",
            Notification.ORDER,
            order,
        )

        remove_ordered_items_from_cart(order)

    else:
        _record(
            order,
            OrderStatus.PENDING_PAYMENT,
            "Awaiting online payment",
        )

    return order


def restock(order, reason):
    """Return stock for every line of an order."""
    items = list(
        order.items.exclude(product=None).order_by("product_id")
    )

    products = {
        product.id: product
        for product in Product.objects.select_for_update()
        .filter(id__in=[item.product_id for item in items])
        .order_by("id")
    }

    for item in items:
        product = products[item.product_id]

        product.stock += item.quantity
        product.sold_count = max(
            0,
            product.sold_count - item.quantity,
        )

        product.save(update_fields=[
            "stock",
            "sold_count",
            "updated_at",
        ])

        log_stock(product, item.quantity, reason, order)


@transaction.atomic
def cancel_order(
    order_id,
    user=None,
    note="Cancelled by customer",
    by=None,
):
    qs = Order.objects.select_for_update()

    if user is not None:
        qs = qs.filter(user=user)

    order = qs.filter(pk=order_id).first()

    if not order:
        raise NotFound("Order not found.")

    if order.status not in CANCELLABLE:
        raise ValidationError({
            "detail": (
                f"An order that is '{order.get_status_display()}' "
                "can no longer be cancelled."
            )
        })

    restock(order, "Order cancelled")

    if order.coupon_id:
        Coupon.objects.filter(
            pk=order.coupon_id,
            used_count__gt=0,
        ).update(used_count=F("used_count") - 1)

    if order.payment_status == PaymentStatus.PAID:
        order.payment_status = PaymentStatus.REFUND_PENDING

    from payments.models import Payment, PaymentRecordStatus

    Payment.objects.filter(
        order=order,
        status=PaymentRecordStatus.CREATED,
    ).update(status=PaymentRecordStatus.CANCELLED)

    order.status = OrderStatus.CANCELLED

    order.save(update_fields=[
        "status",
        "payment_status",
        "updated_at",
    ])

    _record(order, OrderStatus.CANCELLED, note, by)

    message = f"Your order {order.order_number} was cancelled."

    if order.payment_status == PaymentStatus.REFUND_PENDING:
        message += " Your refund will be processed."

    notify(
        order.user,
        "Order cancelled",
        message,
        Notification.ORDER,
        order,
    )

    return order

@transaction.atomic
def advance_status(order_id, new_status, by=None, note=""):
    """Admin-driven forward transition through the delivery flow."""
    order = Order.objects.select_for_update().get(pk=order_id)

    if new_status not in FLOW:
        raise ValidationError({"status": "Invalid target status."})

    if order.status not in FLOW:
        raise ValidationError({
            "status": (
                f"A '{order.get_status_display()}' order "
                "cannot be advanced."
            )
        })

    if FLOW.index(new_status) <= FLOW.index(order.status):
        raise ValidationError({
            "status": (
                "Orders can only move forward through the delivery flow."
            )
        })

    order.status = new_status
    fields = ["status", "updated_at"]

    if (
        new_status == OrderStatus.DELIVERED
        and order.payment_method == PaymentMethod.COD
    ):
        from payments.models import Payment, PaymentRecordStatus

        order.payment_status = PaymentStatus.PAID
        fields.append("payment_status")

        Payment.objects.filter(
            order=order,
            method=PaymentMethod.COD,
        ).update(status=PaymentRecordStatus.SUCCESS)

    order.save(update_fields=fields)

    _record(order, new_status, note, by)

    title, kind = _STATUS_MESSAGES[new_status]

    notify(
        order.user,
        title,
        f"Order {order.order_number}: {OrderStatus(new_status).label}.",
        kind,
        order,
    )

    return order

def build_tracking(order):
    """Timeline for the tracking screen: one entry per stage with completion state and timestamp."""
    reached = {}
    for h in order.history.all():
        reached.setdefault(h.status, h.created_at)
    current = order.status
    cur_idx = FLOW.index(current) if current in FLOW else -1
    stages = [
        {"status": s.value, "label": s.label, "completed": i <= cur_idx, "current": i == cur_idx, "timestamp": reached.get(s.value)}
        for i, s in enumerate(FLOW)
    ]
    return {
        "order_number": order.order_number,
        "status": current,
        "status_label": OrderStatus(current).label,
        "is_cancelled": current == OrderStatus.CANCELLED,
        "awaiting_payment": current == OrderStatus.PENDING_PAYMENT,
        "estimated_delivery": None if current in (OrderStatus.CANCELLED, OrderStatus.DELIVERED) else order.estimated_delivery,
        "stages": stages,
        "history": HistorySerializer(order.history.all(), many=True).data,
    }

"""Invoice PDF generation. render_invoice_pdf() takes plain data so it can be tested without Django."""
from io import BytesIO

from reportlab.lib import colors
from reportlab.lib.pagesizes import A4
from reportlab.lib.styles import getSampleStyleSheet
from reportlab.lib.units import mm
from reportlab.platypus import Paragraph, SimpleDocTemplate, Spacer, Table, TableStyle


def _money(v):
    return f"Rs. {float(v):,.2f}"


def render_invoice_pdf(data):
    """data: order_number, date, customer, shipping_address, payment_method, payment_status,
    items=[{name, qty, unit_price, line_total}], items_total, coupon_code, coupon_discount, delivery_charge, total_amount"""
    buf = BytesIO()
    doc = SimpleDocTemplate(buf, pagesize=A4, leftMargin=18 * mm, rightMargin=18 * mm, topMargin=18 * mm, bottomMargin=18 * mm,
                            title=f"Invoice {data['order_number']}")
    styles = getSampleStyleSheet()
    story = [
        Paragraph("TAX INVOICE", styles["Title"]),
        Paragraph(f"<b>Invoice / Order No:</b> {data['order_number']}<br/><b>Date:</b> {data['date']}", styles["Normal"]),
        Spacer(1, 6 * mm),
        Paragraph(f"<b>Billed to:</b> {data['customer']}<br/>{data['shipping_address']}", styles["Normal"]),
        Spacer(1, 3 * mm),
        Paragraph(f"<b>Payment:</b> {data['payment_method']} ({data['payment_status']})", styles["Normal"]),
        Spacer(1, 6 * mm),
    ]
    rows = [["Item", "Qty", "Unit price", "Amount"]]
    rows += [[Paragraph(i["name"], styles["Normal"]), i["qty"], _money(i["unit_price"]), _money(i["line_total"])] for i in data["items"]]
    rows.append(["", "", "Items total", _money(data["items_total"])])
    if float(data["coupon_discount"]) > 0:
        rows.append(["", "", f"Coupon {data['coupon_code']}", "- " + _money(data["coupon_discount"])])
    rows.append(["", "", "Delivery", _money(data["delivery_charge"])])
    rows.append(["", "", "TOTAL", _money(data["total_amount"])])
    n_items = len(data["items"])
    table = Table(rows, colWidths=[85 * mm, 15 * mm, 35 * mm, 35 * mm], repeatRows=1)
    table.setStyle(TableStyle([
        ("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#222222")),
        ("TEXTCOLOR", (0, 0), (-1, 0), colors.white),
        ("ALIGN", (1, 0), (-1, -1), "RIGHT"),
        ("VALIGN", (0, 0), (-1, -1), "MIDDLE"),
        ("GRID", (0, 0), (-1, n_items), 0.3, colors.grey),
        ("FONTNAME", (2, -1), (-1, -1), "Helvetica-Bold"),
        ("LINEABOVE", (2, n_items + 1), (-1, n_items + 1), 0.5, colors.black),
    ]))
    story += [table, Spacer(1, 10 * mm), Paragraph("Thank you for shopping with us.", styles["Italic"])]
    doc.build(story)
    return buf.getvalue()

import hashlib
import hmac
import json
import logging
from decimal import ROUND_HALF_UP

from django.conf import settings
from django.db import transaction
from rest_framework.exceptions import APIException, NotFound, ValidationError
from cart.services import remove_ordered_items_from_cart
from notifications.models import Notification
from notifications.services import notify

from .models import Payment, PaymentRecordStatus, WebhookEvent

log = logging.getLogger(__name__)


class GatewayError(APIException):
    status_code = 502
    default_detail = "The payment gateway could not be reached. Please try again."


class GatewayNotConfigured(APIException):
    status_code = 503
    default_detail = "Online payments are not configured on the server."


def to_paise(amount):
    return int((amount * 100).to_integral_value(rounding=ROUND_HALF_UP))


def razorpay_client():
    if not (settings.RAZORPAY_KEY_ID and settings.RAZORPAY_KEY_SECRET):
        raise GatewayNotConfigured()
    import razorpay
    return razorpay.Client(auth=(settings.RAZORPAY_KEY_ID, settings.RAZORPAY_KEY_SECRET))


def _hmac_hex(secret, message):
    return hmac.new(secret.encode(), message, hashlib.sha256).hexdigest()


def verify_checkout_signature(razorpay_order_id, razorpay_payment_id, signature):
    """Razorpay checkout signature = HMAC_SHA256(order_id + '|' + payment_id, key_secret)."""
    if not settings.RAZORPAY_KEY_SECRET or not signature:
        return False
    expected = _hmac_hex(settings.RAZORPAY_KEY_SECRET, f"{razorpay_order_id}|{razorpay_payment_id}".encode())
    return hmac.compare_digest(expected, signature)


def verify_webhook_signature(body: bytes, signature):
    """Webhook signature = HMAC_SHA256(raw request body, webhook_secret)."""
    if not settings.RAZORPAY_WEBHOOK_SECRET or not signature:
        return False
    return hmac.compare_digest(_hmac_hex(settings.RAZORPAY_WEBHOOK_SECRET, body), signature)


# ---------------------------------------------------------------- create
@transaction.atomic
def create_payment_order(user, order_id):
    order = Order.objects.select_for_update().filter(pk=order_id, user=user).first()
    if not order:
        raise NotFound("Order not found.")
    if order.payment_method != PaymentMethod.RAZORPAY:
        raise ValidationError({"detail": "This order is not an online-payment order."})
    if order.status != OrderStatus.PENDING_PAYMENT:
        raise ValidationError({"detail": "This order is not awaiting payment."})

    # Retry: reuse the same Razorpay order (Razorpay allows several attempts on one order).
    payment = (
        Payment.objects.select_for_update()
        .filter(order=order, razorpay_order_id__isnull=False, amount=order.total_amount)
        .exclude(status=PaymentRecordStatus.SUCCESS).first()
    )
    if not payment:
        try:
            rp_order = razorpay_client().order.create({
                "amount": to_paise(order.total_amount), "currency": "INR", "receipt": order.order_number,
                "payment_capture": 1, "notes": {"order_id": str(order.pk), "order_number": order.order_number},
            })
        except GatewayNotConfigured:
            raise
        except Exception:
            log.exception("Razorpay order creation failed")
            raise GatewayError()
        payment = Payment.objects.create(
            order=order, method=PaymentMethod.RAZORPAY, amount=order.total_amount, currency="INR",
            razorpay_order_id=rp_order["id"],
        )
    elif payment.status != PaymentRecordStatus.CREATED:
        payment.status = PaymentRecordStatus.CREATED
        payment.failure_reason = ""
        payment.save(update_fields=["status", "failure_reason", "updated_at"])
    if order.payment_status != PaymentStatus.PENDING:
        order.payment_status = PaymentStatus.PENDING
        order.save(update_fields=["payment_status", "updated_at"])

    return {
        "key_id": settings.RAZORPAY_KEY_ID,  # public key id only; the secret never leaves the server
        "razorpay_order_id": payment.razorpay_order_id,
        "amount": to_paise(order.total_amount), "currency": "INR",
        "order_id": order.pk, "order_number": order.order_number,
        "name": "Shop", "description": f"Order {order.order_number}",
        "prefill": {"name": user.first_name, "email": user.email, "contact": user.phone},
    }


# ---------------------------------------------------------------- state changes (caller holds the payment lock)
def _lock_order(payment):
    return Order.objects.select_for_update().get(pk=payment.order_id)


def mark_success(payment, gateway_payment_id, signature="", instrument=""):
    """Idempotent. Returns True only the first time it actually changes state."""
    if payment.status == PaymentRecordStatus.SUCCESS:
        if payment.razorpay_payment_id != gateway_payment_id:
            log.warning("Second payment %s captured for already-paid payment %s - needs manual refund", gateway_payment_id, payment.pk)
        return False
    payment.status = PaymentRecordStatus.SUCCESS
    payment.razorpay_payment_id = gateway_payment_id
    payment.razorpay_signature = signature
    payment.transaction_reference = gateway_payment_id
    payment.instrument = instrument or payment.instrument
    payment.failure_reason = ""
    payment.save()

    order = _lock_order(payment)
    if order.status == OrderStatus.CANCELLED:
        # Money arrived after the order was cancelled/expired: keep the order cancelled, flag for refund.
        order.payment_status = PaymentStatus.REFUND_PENDING
        order.save(update_fields=["payment_status", "updated_at"])
        notify(order.user, "Payment received for a cancelled order",
               f"Your payment for {order.order_number} will be refunded.", Notification.PAYMENT, order)
        return True
    order.payment_status = PaymentStatus.PAID
    fields = ["payment_status", "updated_at"]
    if order.status == OrderStatus.PENDING_PAYMENT:
        order.status = OrderStatus.PLACED
        fields.append("status")
        OrderStatusHistory.objects.create(order=order, status=OrderStatus.PLACED, note="Payment confirmed")
    order.save(update_fields=fields)
    remove_ordered_items_from_cart(order)
    notify(order.user, "Payment successful", f"We received your payment for order {order.order_number}.", Notification.PAYMENT, order)
    notify(order.user, "Order placed", f"Your order {order.order_number} has been placed.", Notification.ORDER, order)
    return True


def mark_failed(payment, reason="Payment failed", status=PaymentRecordStatus.FAILED):
    if payment.status == PaymentRecordStatus.SUCCESS:
        return False  # never downgrade a confirmed payment
    payment.status = status
    payment.failure_reason = reason[:255]
    payment.save(update_fields=["status", "failure_reason", "updated_at"])
    order = _lock_order(payment)
    if order.payment_status == PaymentStatus.PENDING and status == PaymentRecordStatus.FAILED:
        order.payment_status = PaymentStatus.FAILED
        order.save(update_fields=["payment_status", "updated_at"])
        notify(order.user, "Payment failed", f"Payment for order {order.order_number} failed. You can retry.", Notification.PAYMENT, order)
    return True


def apply_gateway_payment(payment, gp, signature=""):
    """Apply a Razorpay payment entity (fetched from the API, or from a signed webhook) to our records.
    Validates order id, amount and currency first. Returns 'success' | 'failed' | 'pending'."""
    if gp.get("order_id") != payment.razorpay_order_id:
        raise ValidationError({"detail": "Payment does not belong to this order."})
    if gp.get("amount") != to_paise(payment.amount) or gp.get("currency", "INR") != payment.currency:
        log.error("Amount mismatch on payment %s: gateway=%s expected=%s", payment.pk, gp.get("amount"), to_paise(payment.amount))
        raise ValidationError({"detail": "Payment amount does not match the order total."})
    state = gp.get("status")
    if state == "captured":
        mark_success(payment, gp["id"], signature, gp.get("method", ""))
        return "success"
    if state == "failed":
        mark_failed(payment, gp.get("error_description") or "Payment failed")
        return "failed"
    return "pending"


# ---------------------------------------------------------------- verify (called by the app after checkout)
def verify_payment(user, razorpay_order_id, razorpay_payment_id, razorpay_signature):
    if not verify_checkout_signature(razorpay_order_id, razorpay_payment_id, razorpay_signature):
        raise ValidationError({"detail": "Invalid payment signature."})
    with transaction.atomic():
        payment = (
            Payment.objects.select_for_update().select_related("order")
            .filter(razorpay_order_id=razorpay_order_id, order__user=user).first()
        )
        if not payment:
            raise NotFound("Payment not found.")
        if payment.status == PaymentRecordStatus.SUCCESS:
            return payment, "success"  # duplicate callback
        try:
            gp = razorpay_client().payment.fetch(razorpay_payment_id)  # amount/status come from Razorpay, not the client
        except GatewayNotConfigured:
            raise
        except Exception:
            log.exception("Razorpay payment fetch failed")
            raise GatewayError()
        outcome = apply_gateway_payment(payment, gp, razorpay_signature)
    payment.refresh_from_db()
    return payment, outcome


@transaction.atomic
def report_client_event(user, razorpay_order_id, outcome, reason=""):
    """The app tells us the customer closed checkout / saw a failure. Informational only:
    it can mark a still-open attempt cancelled/failed but can NEVER mark anything paid."""
    payment = Payment.objects.select_for_update().filter(razorpay_order_id=razorpay_order_id, order__user=user).first()
    if not payment:
        raise NotFound("Payment not found.")
    if payment.status == PaymentRecordStatus.CREATED:
        if outcome == "cancelled":
            mark_failed(payment, "Cancelled by customer", PaymentRecordStatus.CANCELLED)
        else:
            mark_failed(payment, reason or "Payment failed")
    return payment


# ---------------------------------------------------------------- webhook
def handle_webhook(raw_body: bytes, signature, event_id):
    """Returns a short result string. Raises PermissionError on a bad signature."""
    if not verify_webhook_signature(raw_body, signature):
        raise PermissionError("bad signature")
    event = json.loads(raw_body)
    event_id = event_id or hashlib.sha256(raw_body).hexdigest()
    with transaction.atomic():
        record, created = WebhookEvent.objects.get_or_create(
            event_id=event_id, defaults={"event": event.get("event", ""), "payload": event})
        if not created and record.processed:
            return "duplicate"
        entity = (event.get("payload", {}).get("payment") or {}).get("entity")
        note = "ignored"
        if entity and entity.get("order_id") and event.get("event") in ("payment.captured", "order.paid", "payment.failed"):
            payment = Payment.objects.select_for_update().filter(razorpay_order_id=entity["order_id"]).first()
            if not payment:
                note = "unknown order"
            else:
                try:
                    note = apply_gateway_payment(payment, entity)
                except ValidationError as exc:  # mismatch: record it, return 200 so Razorpay stops retrying
                    note = f"rejected: {exc.detail}"[:255]
        record.processed, record.note = True, note
        record.save(update_fields=["processed", "note"])
    return note


# ---------------------------------------------------------------- refunds (admin action)
def refund_order(order):
    """Refund the successful Razorpay payment of an order in full. Returns a message string."""
    payment = order.payments.filter(status=PaymentRecordStatus.SUCCESS, razorpay_payment_id__isnull=False).first()
    if not payment or order.payment_status != PaymentStatus.REFUND_PENDING:
        raise ValidationError("Order has no refund pending.")
    try:
        razorpay_client().payment.refund(payment.razorpay_payment_id, {"amount": to_paise(payment.amount)})
    except Exception as exc:
        log.exception("Refund failed")
        raise GatewayError(f"Refund failed: {exc}")
    order.payment_status = PaymentStatus.REFUNDED
    order.save(update_fields=["payment_status", "updated_at"])
    notify(order.user, "Refund initiated", f"Your refund for order {order.order_number} has been initiated.", Notification.PAYMENT, order)
    return f"Refund initiated for {order.order_number}."

import { useFocusEffect } from '@react-navigation/native';
import { useCallback, useEffect, useRef, useState } from 'react';

interface Options {
  /** Reload silently every time the screen regains focus. */
  refetchOnFocus?: boolean;

  /** Poll interval in ms while the screen is focused. */
  pollMs?: number;

  /** Set false to prevent automatic fetching until enabled. */
  enabled?: boolean;
}

export function useFetch<T>(
  fetcher: () => Promise<T>,
  deps: unknown[] = [],
  opts: Options = {},
) {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const fetcherRef = useRef(fetcher);
  fetcherRef.current = fetcher;

  const mounted = useRef(true);
  const firstFocus = useRef(true);

  const enabled = opts.enabled !== false;

  useEffect(() => {
    return () => {
      mounted.current = false;
    };
  }, []);

  const load = useCallback(
    async (mode: 'initial' | 'refresh' | 'silent' = 'initial') => {
      if (!enabled) return;

      if (mode === 'initial') setLoading(true);
      if (mode === 'refresh') setRefreshing(true);

      try {
        const result = await fetcherRef.current();

        if (!mounted.current) return;

        setData(result);
        setError(null);

      } catch (e) {
        if (mounted.current && mode !== 'silent') {
          setError(
            e instanceof Error
              ? e.message
              : 'Something went wrong',
          );
        }

      } finally {
        if (mounted.current) {
          setLoading(false);
          setRefreshing(false);
        }
      }
    },
    [enabled],
  );

  // Fetch when dependencies change, but only when enabled.
  // eslint-disable-next-line react-hooks/exhaustive-deps
  useEffect(() => {
    if (enabled) {
      load('initial');
    }
  }, [...deps, enabled]);

  useFocusEffect(
    useCallback(() => {
      if (
        enabled &&
        opts.refetchOnFocus &&
        !firstFocus.current
      ) {
        load('silent');
      }

      firstFocus.current = false;

      if (!enabled || !opts.pollMs) {
        return undefined;
      }

      const interval = setInterval(
        () => load('silent'),
        opts.pollMs,
      );

      return () => clearInterval(interval);

    }, [load, enabled, opts.refetchOnFocus, opts.pollMs]),
  );

  return {
    data,
    setData,
    loading,
    refreshing,
    error,
    reload: () => load('initial'),
    refresh: () => load('refresh'),
  };
}

import { paymentApi } from '@/api/services';
import type { RazorpayOrderResponse, User } from '@/types';

export type PayOutcome =
  | { status: 'success' | 'pending'; orderId: number }
  | { status: 'failed' | 'cancelled' | 'unavailable'; orderId: number; reason: string };

function getCheckout(): { open: (o: Record<string, unknown>) => Promise<any> } | null {
  try {
    // Native module: absent in Expo Go, present in a development build.
    const mod = require('react-native-razorpay');
    return mod?.default?.open ? mod.default : null;
  } catch {
    return null;
  }
}

/**
 * Online payment flow (all trust decisions are made by the server):
 * 1. server creates a Razorpay order  2. Razorpay Checkout opens  3. result is sent to /payments/verify/
 * The app never decides a payment succeeded - it only reports what the server confirmed.
 */
export async function payOnline(orderId: number, user: User): Promise<PayOutcome> {
  const checkout = getCheckout();
  if (!checkout) {
    return { status: 'unavailable', orderId, reason: 'Online payment needs the development build of the app (not Expo Go). You can still pay with Cash on Delivery.' };
  }
  let p: RazorpayOrderResponse;
  try {
    p = await paymentApi.createOrder(orderId);
  } catch (e) {
    // e.g. gateway not configured on the server, or the order is no longer awaiting payment
    return { status: 'failed', orderId, reason: e instanceof Error ? e.message : 'Could not start the payment.' };
  }
  const options = {
    key: p.key_id,
    order_id: p.razorpay_order_id,
    amount: p.amount,
    currency: p.currency,
    name: p.name,
    description: p.description,
    prefill: { name: p.prefill.name || user.first_name, email: p.prefill.email, contact: p.prefill.contact },
    theme: { color: '#4F46E5' },
  };
  let result: { razorpay_payment_id: string; razorpay_order_id: string; razorpay_signature: string };
  try {
    result = await checkout.open(options);
  } catch (e: any) {
    const description: string = e?.description ?? e?.error?.description ?? '';
    const cancelled = e?.code === 0 || /cancel/i.test(description);
    await paymentApi.report(p.razorpay_order_id, cancelled ? 'cancelled' : 'failed', description).catch(() => {});
    return { status: cancelled ? 'cancelled' : 'failed', orderId, reason: description || (cancelled ? 'Payment cancelled' : 'Payment failed') };
  }
  try {
    const verified = await paymentApi.verify(result);
    if (verified.status === 'success' || verified.status === 'pending') return { status: verified.status, orderId };
    return { status: 'failed', orderId, reason: 'The bank declined the payment.' };
  } catch (e: any) {
    // Money may have been taken but we could not confirm: tell the user honestly, the webhook can still finalise it.
    return { status: 'pending', orderId };
  }
}

        <View

            styles.paymentIcon,
            selected && styles.paymentIconSelected,
          ]}
        >
          <MaterialCommunityIcons
            name={method.icon as any}

// VELORA CHECKOUT SCREEN
// ======================================================

export default function CheckoutScreen() {
  const nav = useNav();
  const toast = useToast();

  const { user } = useAuth();

  const { refresh: refreshCart } = useCart();

  // ====================================================
  // ADDRESS API
  // ====================================================

  const addresses = useFetch(
    addressApi.list,
    [],
    { refetchOnFocus: true },
  );

  // ====================================================
  // ADDRESS STATE
  // ====================================================

  const [addressId, setAddressId] =
    useState<number | null>(null);

  // ====================================================
  // PAYMENT STATE
  // ====================================================

  const [method, setMethod] =
    useState<PaymentMethod>('RAZORPAY');

  // ====================================================
  // COUPON STATE
  // ====================================================

  const [couponInput, setCouponInput] = useState('');
  const [coupon, setCoupon] = useState('');
  const [couponError, setCouponError] = useState('');
  const [applying, setApplying] = useState(false);

  // ====================================================
  // ORDER STATE
  // ====================================================

  const [placing, setPlacing] = useState(false);

  // ====================================================
  // CHECKOUT SUMMARY API
  // Address and coupon changes refresh the summary.
  // ====================================================

  const summary = useFetch(
    () => orderApi.summary(addressId!, coupon || undefined),
    [addressId, coupon],
    {
      refetchOnFocus: true,
      enabled: addressId !== null,
    },
  );

  // ====================================================
  // AUTO SELECT DEFAULT ADDRESS
  // ====================================================

  useEffect(() => {
    const list = addresses.data;

    if (!list?.length) {
      setAddressId(null);
      return;
    }

    const selectedAddressExists = list.some(
      (address) => address.id === addressId,
    );

    if (addressId === null || !selectedAddressExists) {
      const selectedAddress =
        list.find((address) => address.is_default) ??
        list[0];

      setAddressId(selectedAddress.id);
    }
  }, [addresses.data, addressId]);

  // ====================================================
  // APPLY COUPON
  // ====================================================

  const applyCoupon = async () => {
    if (!couponInput.trim() || applying) return;

    setApplying(true);
    setCouponError('');

    try {
      const response = await orderApi.validateCoupon(
        couponInput.trim(),
      );

      setCoupon(response.code);

      toast.show(
        `Coupon ${response.code} applied`,
        'success',
      );

    } catch (error) {
      setCouponError(errorMessage(error));

    } finally {
      setApplying(false);
    }
  };

  // ====================================================
  // REMOVE COUPON
  // ====================================================

  const removeCoupon = () => {
    setCoupon('');
    setCouponInput('');
    setCouponError('');
  };

  // ====================================================
  // PLACE ORDER
  // ====================================================

  const placeOrder = async () => {
    if (placing) return;

    if (addressId === null) {
      toast.show(
        'Please select a delivery address',
        'error',
      );
      return;
    }

    if (!summary.data) {
      toast.show(
        'Please wait for the order summary.',
        'error',
      );
      return;
    }

    if (!user) {
      toast.show(
        'Please log in to continue.',
        'error',
      );
      return;
    }

    setPlacing(true);

    try {
      // ----------------------------------------------
      // CREATE ORDER
      // Backend recalculates all prices and GST.
      // ----------------------------------------------

      const order = await orderApi.checkout({
        address_id: addressId,
        payment_method: method,
        coupon_code: coupon || undefined,
      });

      // ----------------------------------------------
      // CASH ON DELIVERY
      // ----------------------------------------------

      if (method === 'COD') {
        await refreshCart();

        nav.replace('PaymentResult', {
          status: 'success',
          orderId: order.id,
          method,
        });

        return;
      }

      // ----------------------------------------------
      // RAZORPAY PAYMENT
      // Uses the order created by the backend.
      // ----------------------------------------------

      const outcome = await payOnline(
        order.id,
        user,
      );

      // ----------------------------------------------
      // REFRESH CART AFTER SUCCESS
      // ----------------------------------------------

      if (outcome.status === 'success') {
        await refreshCart();
      }

      // ----------------------------------------------
      // PAYMENT RESULT
      // ----------------------------------------------

      nav.replace('PaymentResult', {
        status: outcome.status,
        orderId: order.id,
        reason:
          'reason' in outcome
            ? outcome.reason
            : undefined,
        method,
      });

    } catch (error) {
      console.error(
        '[CHECKOUT] Order or payment error:',
        error,
      );

      toast.show(
        errorMessage(error),
        'error',
      );

      summary.reload();

    } finally {
      setPlacing(false);
    }
  };

  // ====================================================
  // ADDRESS LOADING
  // ====================================================

  if (addresses.loading) {
    return <LoadingIndicator />;
  }

  // ====================================================
  // ADDRESS ERROR
  // ====================================================

  if (addresses.error) {
    return (
      <ErrorMessage
        message={addresses.error}
        onRetry={addresses.reload}
      />
    );
  }

  // ====================================================
  // SUMMARY LOADING
  // Only load after selecting an address.
  // ====================================================

  if (addressId !== null && summary.loading) {
    return <LoadingIndicator />;
  }

  // ====================================================
  // SUMMARY ERROR
  // ====================================================

  if (addressId !== null && summary.error) {
    return (
      <ErrorMessage
        message={summary.error}
        onRetry={summary.reload}
      />
    );
  }

  // ====================================================
  // CHECKOUT DATA
  // ====================================================

  const addressList = addresses.data ?? [];

  const s = summary.data;

  // ====================================================
  // GST BREAKDOWN ROWS
  // ====================================================

  const gstRows = s
    ? s.tax_type === 'INTRA_STATE'
      ? [
          {
            label: 'CGST',
            value: s.cgst_amount,
          },
          {
            label: 'SGST',
            value: s.sgst_amount,
          },
        ]
      : [
          {
            label: 'IGST',
            value: s.igst_amount,
          },
        ]
    : [];

  // ====================================================
  // ORDER PRICE BREAKDOWN
  // ====================================================

  const priceRows = s
    ? [
        {
          label: `Items (${s.item_count})`,
          value: s.original_items_total,
        },

        ...(Number(s.product_discount) > 0
          ? [
              {
                label: 'Product Discount',
                value: s.product_discount,
                negative: true,
              },
            ]
          : []),

        ...(Number(s.coupon_discount) > 0
          ? [
              {
                label: `Coupon ${s.coupon_code}`,
                value: s.coupon_discount,
                negative: true,
              },
            ]
          : []),

        {
          label: 'Delivery',
          value: s.delivery_charge,
        },

        {
          label: 'Taxable Amount',
          value: s.taxable_amount,
        },

        ...gstRows,

        {
          label: 'Total Payable',
          value: s.total_amount,
          bold: true,
        },
      ]
    : [];

  // ====================================================
  // MAIN UI
  // ====================================================

  return (
    <View style={styles.container}>

      <ScrollView
        showsVerticalScrollIndicator={false}
        contentContainerStyle={styles.scrollContent}
        keyboardShouldPersistTaps="handled"
      >

        {/* CHECKOUT HEADER */}

        <Animated.View
          entering={FadeInDown.duration(450)}
          style={styles.pageHeader}
        >
          <View style={styles.pageHeaderIcon}>
            <MaterialCommunityIcons
              name="cart-check"
              size={27}
              color={colors.primary}
            />
          </View>

          <View style={styles.pageHeaderText}>
            <Text style={styles.pageTitle}>
              Secure Checkout
            </Text>

            <Text style={styles.pageSubtitle}>
              Complete your order with VELORA.
            </Text>
          </View>
        </Animated.View>

        {/* STEP 1 — DELIVERY ADDRESS */}

        <SectionTitle
          n={1}
          title="Delivery Address"
          subtitle="Where should we deliver your order?"
        />

        {addressList.length > 0 ? (
          <View style={styles.addressList}>
            {addressList.map((address, index) => (
              <Animated.View
                key={address.id}
                entering={FadeInDown
                  .delay(index * 75)
                  .duration(400)}
              >
                <AddressCard
                  address={address}
                  selected={address.id === addressId}
                  onPress={() => setAddressId(address.id)}
                />
              </Animated.View>
            ))}
          </View>
        ) : (
          <EmptyState
            icon="map-marker-plus-outline"
            title="No Saved Address"
            subtitle="Add your delivery address to continue."
          />
        )}

        <View style={styles.addAddressContainer}>
          <CustomButton
            label="Add New Address"
            mode="outlined"
            icon="plus"
            onPress={() => nav.navigate('AddressForm')}
            style={styles.addAddressButton}
          />
        </View>

        {/* STEP 2 — PAYMENT METHOD */}

        <SectionTitle
          n={2}
          title="Payment Method"
          subtitle="Choose how you want to pay."
        />

        {METHODS.map((payment, index) => (
          <PaymentMethodCard
            key={payment.value}
            method={payment}
            index={index}
            selected={method === payment.value}
            onPress={() => setMethod(payment.value)}
          />
        ))}

        {/* STEP 3 — COUPON */}

        <SectionTitle
          n={3}
          title="Apply Coupon"
          subtitle="Have a discount code? Use it here."
        />

        {coupon ? (
          <Animated.View
            entering={FadeInDown.duration(350)}
            style={styles.appliedCoupon}

            <View style={styles.couponIcon}>
              <MaterialCommunityIcons
                name="ticket-percent"
                size={25}
                color={colors.success}
              />

            <View style={styles.appliedCouponText}>
              <Text style={styles.appliedCouponCode}>

              <Text style={styles.appliedCouponSubtitle}>
                Coupon applied successfully
              </Text>

            <Pressable
              onPress={removeCoupon}
              hitSlop={10}
              accessibilityRole="button"
              accessibilityLabel="Remove coupon"
            >
              <MaterialCommunityIcons
                name="close-circle"
                size={25}
                color={colors.danger}
              />

          </Animated.View>
        ) : (
          <Animated.View
            entering={FadeInDown.duration(350)}

            <View style={styles.couponInputContainer}>
              <TextInput
                mode="outlined"
                label="Enter Coupon Code"

                  setCouponInput(text.toUpperCase());

                autoCapitalize="characters"

              size={30}

      </Animated.View>

    </View>
  );
}

import React from 'react';
import { Text, View } from 'react-native';
import { colors } from '@/constants/theme';
import { formatINR } from '@/utils/format';

interface Row {
  label: string;
  value: string | number;
  negative?: boolean;
  free?: boolean;
  bold?: boolean;
}

export default function PriceSummary({ rows }: { rows: Row[] }) {
  return (
    <View style={{ backgroundColor: '#fff', borderRadius: 12, padding: 14, borderWidth: 1, borderColor: colors.border }}>
      {rows.map((r, i) => (
        <View key={r.label} style={{ flexDirection: 'row', justifyContent: 'space-between', paddingVertical: 4, ...(r.bold && i > 0 ? { borderTopWidth: 1, borderTopColor: colors.border, marginTop: 6, paddingTop: 10 } : {}) }}>
          <Text style={{ color: r.bold ? colors.text : colors.muted, fontWeight: r.bold ? '700' : '400', fontSize: r.bold ? 16 : 14 }}>{r.label}</Text>
          <Text style={{ color: r.negative || r.free ? colors.success : colors.text, fontWeight: r.bold ? '700' : '500', fontSize: r.bold ? 16 : 14 }}>
            {r.free && Number(r.value) === 0 ? 'FREE' : `${r.negative ? '- ' : ''}${formatINR(r.value)}`}
          </Text>
        </View>
      ))}
    </View>
  );
}

export const formatINR = (value: string | number | null | undefined): string => {
  const n = Number(value ?? 0);
  return '₹' + n.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
};

export const formatDate = (iso?: string | null): string => {
  if (!iso) return '';
  const d = new Date(iso);
  return d.toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' });
};

export const formatDateTime = (iso?: string | null): string => {
  if (!iso) return '';
  const d = new Date(iso);
  return `${formatDate(iso)}, ${d.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })}`;
};
